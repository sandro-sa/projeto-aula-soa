2. Entenda o que será feito

O teste publica os descritores em Tomcat temporário, consome os proxies e confere o QR gerado.

  • As execuções automáticas usam H2 em modo MySQL e portas de teste.
  • O teste opt-in do MySQL precisa de configuração própria; ignorado não significa aprovado.
Pontos para observar no código
  • assertThrows verifica a recusa; asserções de código evitam aprovar uma exceção inesperada.
  • O teste de rollback insere e lança erro antes de commit; a consulta deve mostrar que não persistiu.
  • O teste de preço histórico muda cadastro entre inclusões e verifica linhas independentes.

package br.edu.soa.restaurante;

import br.edu.soa.restaurante.soap.*;
import jakarta.xml.ws.*;
import javax.xml.namespace.QName;
import org.apache.catalina.startup.Tomcat;
import org.apache.catalina.webresources.*;
import org.junit.jupiter.api.*;
import org.junit.jupiter.api.io.TempDir;
import static org.junit.jupiter.api.Assertions.*;
import java.nio.file.*;
import java.net.*;
import java.net.http.*;
import java.util.*;
import java.math.BigDecimal;

/** Publica os descritores reais no Tomcat 10.1 e usa XML/HTTP + proxies do WSDL. */
class SoapHttpTest {
    @TempDir static Path pasta;
    static Tomcat tomcat; static String base;
    static final String NS="http://restaurante.soa.edu.br/v2";
    static final String SENHA="teste-admin-123456", GARCOM="teste-garcom-123456";
    @BeforeAll static void iniciar() throws Exception {
        Map.of("DB_URL","jdbc:h2:mem:soap;MODE=MySQL;DB_CLOSE_DELAY=-1","DB_USER","sa","DB_PASSWORD","",
            "RESTAURANTE_ADMIN_PASSWORD",SENHA,"RESTAURANTE_GARCOM_PASSWORD",GARCOM).forEach((k,v) -> System.setProperty("restaurante."+k,v));
        Path app=pasta.resolve("app");Files.createDirectories(app.resolve("WEB-INF/wsdl"));
        for(String f:List.of("web.xml","sun-jaxws.xml")) Files.copy(Path.of("src/main/webapp/WEB-INF",f),app.resolve("WEB-INF").resolve(f));
        try(var files=Files.list(Path.of("../contrato/src/main/resources/wsdl"))) {for(Path f:files.toList())Files.copy(f,app.resolve("WEB-INF/wsdl").resolve(f.getFileName()));}
        Files.createDirectories(app.resolve("cardapio"));Files.copy(Path.of("../../../cardapio-web/index.html"),app.resolve("cardapio/index.html"));
        tomcat=new Tomcat();tomcat.setBaseDir(pasta.resolve("tomcat").toString());tomcat.setPort(0);
        tomcat.setAddDefaultWebXmlToWebapp(false);
        tomcat.getConnector().setProperty("address","127.0.0.1");
        var contexto=tomcat.addWebapp("/restaurante",app.toAbsolutePath().toString());contexto.setParentClassLoader(SoapHttpTest.class.getClassLoader());
        ((org.apache.catalina.core.StandardContext)contexto).setDelegate(true);
        Tomcat.addServlet(contexto,"default",new org.apache.catalina.servlets.DefaultServlet()).setLoadOnStartup(1);
        contexto.addServletMappingDecoded("/","default");Tomcat.addDefaultMimeTypeMappings(contexto);
        var recursos=new StandardRoot(contexto);
        recursos.addPreResources(new DirResourceSet(recursos,"/WEB-INF/classes",Path.of("target/classes").toAbsolutePath().toString(),"/"));
        contexto.setResources(recursos);tomcat.start();
        assertTrue(contexto.getState().isAvailable(),"WAR deve inicializar");
        base="http://127.0.0.1:"+tomcat.getConnector().getLocalPort()+"/restaurante";
    }
    @AfterAll static void parar() throws Exception {
        if(tomcat!=null) {tomcat.stop();tomcat.destroy();}
        for(String k:List.of("DB_URL","DB_USER","DB_PASSWORD","RESTAURANTE_ADMIN_PASSWORD","RESTAURANTE_GARCOM_PASSWORD"))System.clearProperty("restaurante."+k);
    }
    static <T> T port(String entidade,Class<T> tipo,String usuario) {
        var service=Service.create(SoapHttpTest.class.getResource("/wsdl/"+entidade+"Service.wsdl"),new QName(NS,entidade+"Service"));
        T proxy=service.getPort(new QName(NS,entidade+"Port"),tipo);var c=((BindingProvider)proxy).getRequestContext();
        c.put(BindingProvider.ENDPOINT_ADDRESS_PROPERTY,base+"/soap/"+entidade.toLowerCase()+"s");
        c.put(BindingProvider.USERNAME_PROPERTY,usuario);c.put(BindingProvider.PASSWORD_PROPERTY,"admin".equals(usuario)?SENHA:GARCOM);return proxy;
    }
    // Cenário: quatro wsdl publicados eprotegidos.
    @Test void quatroWsdlPublicadosEProtegidos() throws Exception {
        HttpClient http=HttpClient.newHttpClient();String basic=Base64.getEncoder().encodeToString(("admin:"+SENHA).getBytes(java.nio.charset.StandardCharsets.UTF_8));
        for(String recurso:List.of("mesas","categorias","produtos","pedidos")) {
            URI uri=URI.create(base+"/soap/"+recurso+"?wsdl");
            assertEquals(401,http.send(HttpRequest.newBuilder(uri).build(),HttpResponse.BodyHandlers.ofString()).statusCode());
            var resposta=http.send(HttpRequest.newBuilder(uri).header("Authorization","Basic "+basic).build(),HttpResponse.BodyHandlers.ofString());
            assertEquals(200,resposta.statusCode());assertTrue(resposta.body().contains("portType"));
        }
    }
    // Cenário: fluxo soap completo efault.
    @Test void fluxoSoapCompletoEFAULT() throws Exception {
        var mesas=port("Mesa",MesaServicePortType.class,"admin");var cats=port("Categoria",CategoriaServicePortType.class,"admin");var prods=port("Produto",ProdutoServicePortType.class,"admin");var pedidos=port("Pedido",PedidoServicePortType.class,"garcom");
        MesaEntrada me=new MesaEntrada();me.setNumero(2);var m=mesas.criarMesa(me);
        assertEquals("DUPLICADO",assertThrows(ErroNegocio_Exception.class,() -> mesas.criarMesa(me)).getFaultInfo().getCodigo());
        CategoriaEntrada ce=new CategoriaEntrada();ce.setNome("Massas");ce.setDescricao("Artesanais");ce.setAtivo(true);var cat=cats.criarCategoria(ce);
        ProdutoEntrada pe=new ProdutoEntrada();pe.setCategoriaId(cat.getId());pe.setNome("Carbonara");pe.setDescricao("Ovos e queijo");pe.setImagem("");pe.setPreco(new BigDecimal("55.00"));pe.setPorcao("1 pessoa");pe.setDisponivel(true);var prod=prods.criarProduto(pe);
        var p=pedidos.abrirPedido(m.getId());p=pedidos.adicionarItem(p.getId(),prod.getId(),2);assertEquals(new BigDecimal("110.00"),p.getTotal());
        long id=p.getId();long item=p.getItens().get(0).getId();
        assertEquals(new BigDecimal("165.00"),pedidos.atualizarQuantidadeItem(id,item,3).getTotal());
        assertTrue(pedidos.removerItem(id,item).getItens().isEmpty());pedidos.adicionarItem(id,prod.getId(),2);
        assertEquals(SituacaoPedido.FECHADO,pedidos.fecharPedido(id).getSituacao());
        assertEquals("PEDIDO_FECHADO",assertThrows(ErroNegocio_Exception.class,() -> pedidos.adicionarItem(id,prod.getId(),1)).getFaultInfo().getCodigo());
        assertEquals(SituacaoMesa.LIVRE,mesas.consultarMesa(m.getId()).getSituacao());
        assertEquals("ACESSO_NEGADO",assertThrows(ErroNegocio_Exception.class,() -> port("Mesa",MesaServicePortType.class,"garcom").criarMesa(me)).getFaultInfo().getCodigo());
    }
    // Cenário: bff publico nao expoe pedidos.
    @Test void bffPublicoNaoExpoePedidos() throws Exception {
        var http=HttpClient.newHttpClient();var r=http.send(HttpRequest.newBuilder(URI.create(base+"/api/public/cardapio")).build(),HttpResponse.BodyHandlers.ofString());
        assertEquals(200,r.statusCode());assertTrue(r.body().contains("categorias"));assertFalse(r.body().contains("mesaId"));
        assertEquals(404,http.send(HttpRequest.newBuilder(URI.create(base+"/api/public/pedidos")).build(),HttpResponse.BodyHandlers.ofString()).statusCode());
        assertEquals(200,http.send(HttpRequest.newBuilder(URI.create(base+"/cardapio/index.html")).build(),HttpResponse.BodyHandlers.ofString()).statusCode());
    }
    // Cenário: cliente usado pelo swing consome servidor.
    @Test void clienteUsadoPeloSwingConsomeServidor() throws Exception {
        String anterior=System.getProperty("soap.base");System.setProperty("soap.base",base);
        try {
            ConexaoSoap cliente=new ConexaoSoap("admin",SENHA);
            MesaEntrada entrada=new MesaEntrada();entrada.setNumero(12);
            Mesa m=cliente.mesas().criarMesa(entrada);
            assertEquals(12,cliente.mesas().consultarMesa(m.getId()).getNumero());
        } finally {if(anterior==null)System.clearProperty("soap.base");else System.setProperty("soap.base",anterior);}
    }
}

package br.edu.soa.restaurante.infraestrutura;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;
import javax.imageio.ImageIO;
import java.io.ByteArrayInputStream;
import com.google.zxing.*;
import com.google.zxing.common.HybridBinarizer;
import com.google.zxing.client.j2se.BufferedImageLuminanceSource;
import br.edu.soa.restaurante.aplicacao.Falha;

class QrCodeTest {
    // Cenário: qr contem url real enao cardapio.
    @Test void qrContemUrlRealENaoCardapio() throws Exception {
        var img=ImageIO.read(new ByteArrayInputStream(QrCode.gerar("https://restaurante.exemplo/restaurante",2)));
        var texto=new MultiFormatReader().decode(new BinaryBitmap(new HybridBinarizer(new BufferedImageLuminanceSource(img)))).getText();
        assertEquals("https://restaurante.exemplo/restaurante/cardapio/index.html?mesa=2",texto);
    }
    // Cenário: rejeita configuracao invalida.
    @Test void rejeitaConfiguracaoInvalida() {
        assertThrows(Falha.class,() -> QrCode.gerar(null,2));assertThrows(Falha.class,() -> QrCode.gerar("file:///segredo",2));assertThrows(Falha.class,() -> QrCode.gerar("https://exemplo.org",0));
    }
}

Se a verificação não der certo
  • Leia a mensagem específica na Console ou na aba Problems; confira o projeto, o caminho do arquivo e o pacote.
  • Depois de mudar POMs, use Maven → Update Project…. Confira se cada módulo listado já possui seu POM.
