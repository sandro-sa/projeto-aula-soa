2. Entenda o que será feito

Problema no sistema

O contrato HTTP precisa coincidir com as rotas que o servidor executa. A suíte exercita autorização, regras e status e salva a descrição OpenAPI gerada para conferência.

MockMvc percorre MVC e Security com dados de teste sem abrir uma porta HTTP externa.

  • O teste inicial recebe configuração porque a aplicação agora exige banco e credenciais.
  • Inspecione os relatórios e os status de validação, autorização e conflito.
Pontos para observar no código
  • O teste básico recebeu configuração porque a aplicação evoluiu desde o bootstrap vazio.
  • Os testes HTTP conferem 200/201/400/401/403/404/405/409/415.
  • A verificação OpenAPI exige schema de erro correto e resposta de sucesso nas operações.

// Finalidade: Verificar os cenários desta etapa por asserções automatizadas.
package br.edu.soa.restaurante.rest;

import org.junit.jupiter.api.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.test.web.servlet.*;
import tools.jackson.databind.*;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.*;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;
import static org.springframework.security.test.web.servlet.request.SecurityMockMvcRequestPostProcessors.httpBasic;
import static org.junit.jupiter.api.Assertions.*;
import java.nio.file.*;

@SpringBootTest(properties={"restaurante.db-url=jdbc:h2:mem:rest;MODE=MySQL;DB_CLOSE_DELAY=-1","restaurante.db-user=sa","restaurante.db-password=","restaurante.admin-password=teste-admin-123456","restaurante.garcom-password=teste-garcom-123456"})
@AutoConfigureMockMvc
class RestHttpTest {
    @Autowired MockMvc mvc; @Autowired ObjectMapper json;
    static final String ADMIN="teste-admin-123456",GARCOM="teste-garcom-123456";
    JsonNode criar(String recurso,String corpo) throws Exception {
        return json.readTree(mvc.perform(post("/api/"+recurso).with(httpBasic("admin",ADMIN)).contentType("application/json").content(corpo)).andExpect(status().isCreated()).andExpect(header().exists("Location")).andReturn().getResponse().getContentAsString());
    }
    // Cenário: fluxo rest com calculos eerros.
    @Test void fluxoRestComCalculosEErros() throws Exception {
        long mesa=criar("mesas","{\"numero\":2}").get("id").asLong();
        mvc.perform(post("/api/mesas").with(httpBasic("admin",ADMIN)).contentType("application/json").content("{\"numero\":2}")).andExpect(status().isConflict()).andExpect(jsonPath("$.codigo").value("DUPLICADO"));
        long categoria=criar("categorias","{\"nome\":\"Massas\",\"descricao\":\"Artesanais\",\"ativo\":true}").get("id").asLong();
        long produto=criar("produtos","{\"categoriaId\":"+categoria+",\"nome\":\"Carbonara\",\"descricao\":\"Ovos\",\"imagem\":\"\",\"preco\":55.00,\"porcao\":\"1 pessoa\",\"disponivel\":true}").get("id").asLong();
        long pedido=criar("pedidos","{\"mesaId\":"+mesa+"}").get("id").asLong();
        String url="/api/pedidos/"+pedido;
        var p=json.readTree(mvc.perform(post(url+"/itens").with(httpBasic("garcom",GARCOM)).contentType("application/json").content("{\"produtoId\":"+produto+",\"quantidade\":2}")).andExpect(status().isCreated()).andExpect(jsonPath("$.total").value(110.00)).andReturn().getResponse().getContentAsString());
        long item=p.get("itens").get(0).get("id").asLong();
        mvc.perform(patch(url+"/itens/"+item).with(httpBasic("garcom",GARCOM)).contentType("application/json").content("{\"quantidade\":3}")).andExpect(status().isOk()).andExpect(jsonPath("$.total").value(165.00));
        mvc.perform(patch(url+"/itens/"+item).with(httpBasic("garcom",GARCOM)).contentType("application/json").content("{\"quantidade\":0}")).andExpect(status().isBadRequest());
        mvc.perform(delete(url+"/itens/"+item).with(httpBasic("garcom",GARCOM))).andExpect(status().isOk()).andExpect(jsonPath("$.total").value(0));
        mvc.perform(post(url+"/itens").with(httpBasic("garcom",GARCOM)).contentType("application/json").content("{\"produtoId\":"+produto+",\"quantidade\":2,\"total\":1}")).andExpect(status().isBadRequest());
        mvc.perform(post(url+"/itens").with(httpBasic("garcom",GARCOM)).contentType("application/json").content("{\"produtoId\":"+produto+",\"quantidade\":2}")).andExpect(status().isCreated());
        mvc.perform(post(url+"/fechamento").with(httpBasic("garcom",GARCOM))).andExpect(status().isOk()).andExpect(jsonPath("$.situacao").value("FECHADO"));
        mvc.perform(delete(url+"/itens/"+item).with(httpBasic("garcom",GARCOM))).andExpect(status().isConflict()).andExpect(jsonPath("$.codigo").value("PEDIDO_FECHADO"));
        mvc.perform(get("/api/mesas/"+mesa).with(httpBasic("garcom",GARCOM))).andExpect(jsonPath("$.situacao").value("LIVRE"));
        mvc.perform(get("/api/mesas/99999").with(httpBasic("garcom",GARCOM))).andExpect(status().isNotFound());
        mvc.perform(get("/api/produtos?limite=101").with(httpBasic("admin",ADMIN))).andExpect(status().isBadRequest());
        mvc.perform(post("/api/mesas").with(httpBasic("admin",ADMIN)).contentType("text/plain").content("mesa")).andExpect(status().isUnsupportedMediaType());
        mvc.perform(patch("/api/mesas/"+mesa).with(httpBasic("admin",ADMIN))).andExpect(status().isMethodNotAllowed());
    }
    // Cenário: acesso publico eperfis.
    @Test void acessoPublicoEPerfis() throws Exception {
        mvc.perform(get("/api/public/cardapio?mesa=999")).andExpect(status().isOk()).andExpect(jsonPath("$.produtos").isArray()).andExpect(jsonPath("$.pedidos").doesNotExist());
        mvc.perform(get("/api/pedidos")).andExpect(status().isUnauthorized());
        mvc.perform(post("/api/mesas").with(httpBasic("garcom",GARCOM)).contentType("application/json").content("{\"numero\":9}")).andExpect(status().isForbidden());
        mvc.perform(get("/cardapio/index.html")).andExpect(status().isOk());
        mvc.perform(get("/swagger-ui/index.html").with(httpBasic("admin",ADMIN))).andExpect(status().isOk());
    }
    // Cenário: exportar contrato open api.
    @Test void exportarContratoOpenApi() throws Exception {
        String contrato=mvc.perform(get("/v3/api-docs").with(httpBasic("admin",ADMIN))).andExpect(status().isOk()).andReturn().getResponse().getContentAsString();
        JsonNode spec=json.readTree(contrato);assertEquals("3.0.1",spec.get("openapi").asText());
        for(String p:new String[]{"/api/mesas","/api/categorias","/api/produtos","/api/pedidos","/api/pedidos/{id}/itens","/api/pedidos/{id}/fechamento"})assertTrue(spec.get("paths").has(p),p);
        assertTrue(spec.get("components").get("schemas").has("PedidoDto"));
        for(String p:new String[]{"/api/mesas","/api/categorias","/api/produtos","/api/pedidos"}) {
            var respostas=spec.get("paths").get(p).get("get").get("responses");
            assertTrue(respostas.has("200"));
            for(String code:new String[]{"400","401","403","404","409","503"})
                assertEquals("#/components/schemas/Erro",respostas.get(code).get("content").get("application/json").get("schema").get("$ref").asText());
        }
        assertEquals("/",spec.get("servers").get(0).get("url").asText());
        Files.createDirectories(Path.of("target"));Files.writeString(Path.of("target/openapi.json"),json.writerWithDefaultPrettyPrinter().writeValueAsString(spec));
    }
}

  • Leia a mensagem específica na Console ou na aba Problems; confira o projeto, o caminho do arquivo e o pacote.
  • Depois de mudar POMs, use Maven → Update Project…. Confira se cada módulo listado já possui seu POM.
  • No Maven build, confira Base directory, Goals e o JDK 17. Em erro de download, confira a conexão e o repositório informado.
  • Em problema de execução, diferencie configuração do processo, conexão com o banco, rota/contrato e regra de negócio. Consulte a aula de diagnóstico.
