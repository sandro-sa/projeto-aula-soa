2. Entenda o que será feito

O adaptador relaciona tipos de falha aos status e mantém um corpo de erro previsível.

  • Erros de validação de DTO e Falha da aplicação chegam por caminhos diferentes.
  • A resposta pública não inclui stacktrace ou detalhe de SQL.
Pontos para observar no código
  • As responsabilidades ficam neste arquivo; o domínio continua sem anotações de framework.
  • Tipos e dependências referenciados foram introduzidos nas aulas anteriores.
  • Uma exceção da regra será transformada na borda, mantendo a mesma regra do SOAP.

// Finalidade: Traduzir exceções para status HTTP e um corpo de erro consistente.
package br.edu.soa.restaurante.rest;

import br.edu.soa.restaurante.aplicacao.Falha;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;
import jakarta.validation.ConstraintViolationException;

@RestControllerAdvice
public class Erros {
    public record Erro(
        @io.swagger.v3.oas.annotations.media.Schema(example="PEDIDO_FECHADO") String codigo,
        @io.swagger.v3.oas.annotations.media.Schema(example="Pedido fechado não permite alterações.") String mensagem) {}
    @ExceptionHandler(Falha.class)
    public ResponseEntity<Erro> negocio(Falha e) {
        int status=switch(e.tipo()) { case VALIDACAO -> 400; case NAO_ENCONTRADO -> 404; case CONFLITO -> 409; case INTERNO -> 503; };
        return ResponseEntity.status(status).body(new Erro(e.codigo(),e.getMessage()));
    }
    @ExceptionHandler({MethodArgumentNotValidException.class,HttpMessageNotReadableException.class,MethodArgumentTypeMismatchException.class,ConstraintViolationException.class})
    public ResponseEntity<Erro> entrada(Exception e) { return ResponseEntity.badRequest().body(new Erro("ENTRADA_INVALIDA","Verifique os campos, tipos e limites da requisição.")); }
    @ExceptionHandler(org.springframework.web.HttpRequestMethodNotSupportedException.class)
    public ResponseEntity<Erro> metodo(Exception e) { return ResponseEntity.status(405).body(new Erro("METODO_INVALIDO","Método HTTP não permitido neste recurso.")); }
    @ExceptionHandler(org.springframework.web.HttpMediaTypeNotSupportedException.class)
    public ResponseEntity<Erro> formato(Exception e) { return ResponseEntity.status(415).body(new Erro("FORMATO_INVALIDO","Envie o corpo com Content-Type application/json.")); }
    @ExceptionHandler(org.springframework.web.servlet.resource.NoResourceFoundException.class)
    public ResponseEntity<Erro> inexistente(Exception e) { return ResponseEntity.status(404).body(new Erro("NAO_ENCONTRADO","Recurso não encontrado.")); }
    @ExceptionHandler(Exception.class)
    public ResponseEntity<Erro> interno(Exception e) {
        System.getLogger(Erros.class.getName()).log(System.Logger.Level.ERROR,"Erro interno REST: {0}",e.getClass().getSimpleName());
        return ResponseEntity.status(500).body(new Erro("INTERNO","Não foi possível concluir a operação."));
    }
}
